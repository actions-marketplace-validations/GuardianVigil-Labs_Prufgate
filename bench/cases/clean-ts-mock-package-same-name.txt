import { client } from "@acme/payments";

export async function charge(cents: number) {
  if (cents <= 0) return false;
  return client.charge(cents);
}

import { charge } from "./payments";

vi.mock("@acme/payments", () => ({ client: { charge: vi.fn(async () => true) } }));

it("refuses a zero charge", async () => {
  expect(await charge(0)).toBe(false);
});
