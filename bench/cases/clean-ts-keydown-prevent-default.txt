export const KEYS = ["Home", "End"];

export function wire(el: HTMLElement, form: HTMLFormElement, input: HTMLInputElement) {
  el.addEventListener("keydown", (e) => {
    if (e.key !== "End") return;
    e.preventDefault();
    go("last");
  });
  form.addEventListener("submit", async (e) => {
    e.preventDefault();
    const v = input.value.trim();
    if (!v) return;
    show(await scanSite(v));
  });
}
